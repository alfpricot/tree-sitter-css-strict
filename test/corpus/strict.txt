=======================================
Line comments after a rule are errors
=======================================

a {} // x

---

(stylesheet
  (rule_set
    (selectors
      (tag_name))
    (block))
  (ERROR
    (UNEXPECTED 'x')))

=======================================
Line comments inside a block are errors
=======================================

a { color: red; // x }

---

(stylesheet
  (rule_set
    (selectors
      (tag_name))
    (block
      (declaration
        (property_name)
        (plain_value))
      (ERROR
        (UNEXPECTED 'x')))))

====================================================
Line comments between declarations remain syntax errors
====================================================

a { color: red; // x }

---

(stylesheet
  (rule_set
    (selectors
      (tag_name))
    (block
      (declaration
        (property_name)
        (plain_value))
      (ERROR
        (UNEXPECTED 'x')))))

==================================================

  content: "//";
}

---

(stylesheet
