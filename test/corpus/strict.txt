=======================================
Line comments after a rule are errors
=======================================

a {} // x

---

(stylesheet
  (rule_set
    (selectors
      (tag_name))
    (block))
  (ERROR
    (UNEXPECTED 'x')))

=======================================
Line comments inside a block are errors
=======================================

---

(stylesheet
  (rule_set
    (selectors
      (tag_name))
    (block
      (declaration
        (property_name)
        (plain_value))
      (ERROR
        (UNEXPECTED 'x')))))

  width: calc(.5em + 2px);
  margin: max(.5rem, 1px);
  transform: scale(.5);
  src: url(./a.png), url(.5x/bg.png), url(.hidden/b.png);
}

---

(stylesheet
